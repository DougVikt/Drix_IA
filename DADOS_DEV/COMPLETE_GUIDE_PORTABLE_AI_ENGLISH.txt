 PHASE 2 - PORTABLE PYTHON INSTALLATION (WinPython)
================================================================================

  Goal: Have a Python interpreter that runs entirely from the HD without
        touching the Windows Registry of the host PC.

  Steps:

  1. Download WinPython 3.10.11.0dot (Zero version) from the official
     GitHub releases page.

  2. Run the installer and set destination to: [DRIVE]:\ia_system\python\
     Example: E:\ia_system\python\

  3. Select "Extract" and wait for completion.

  4. After extraction, DO NOT rename the internal folder.

  5. To verify, open "WinPython Command Prompt.exe" inside the extracted
     folder and type:
       python --version
     It should return: Python 3.10.11

  6. Still in the WinPython terminal, upgrade pip:
       python -m pip install --upgrade pip

  NOTE: Never use the regular Windows CMD to install libraries; always use
        "WinPython Command Prompt.exe" to ensure packages go to the HD
        and not to the host PC.

 PHASE 3 - POETRY SETUP (PACKAGE MANAGER)
================================================================================

  Poetry is superior to pip because:
  - Guarantees library versions never conflict
  - Creates a poetry.lock file that "locks" versions
  - Rebuilds the environment on any PC with poetry install

  3.1 Install Poetry inside WinPython (on the HD)

    Open WinPython Command Prompt.exe and run:

      pip install poetry

  3.2 Configure Portability (CRITICAL)

    By default, Poetry saves virtual environments in the host PC's AppData.
    Run these commands in the WinPython terminal to force the environment
    to stay on the HD:

      poetry config virtualenvs.in-project true
      poetry config cache-dir "./ia_system/poetry_cache"

  3.3 Initialize the Project

    In the WinPython terminal, navigate to the HD root and run:

      cd [DRIVE]:\
      poetry init

    Follow the interactive prompts:
      - Package name: portable-ai
      - Version: 0.1.0
      - Description: Portable Independent AI with Continuous Learning
      - Author: (your name)
      - License: MIT
      - Define dependencies interactively? Yes

  3.4 Add Core Dependencies

    Still in the WinPython terminal, at the HD root:

      poetry add langchain langchain-community langchain-ollama langchain-chroma langchain-text-splitters chromadb ollama openai pyyaml fastapi uvicorn streamlit

    This creates poetry.lock. Keep it safe - it's your reproducibility
    guarantee.

  3.5 Full pyproject.toml (for reference)

import os
import sys
import yaml
import importlib
from pathlib import Path

# Load configuration
CONFIG_PATH = Path(__file__).parent / "config.yaml"
with open(CONFIG_PATH, "r", encoding="utf-8") as f:
    config = yaml.safe_load(f)

# Auto-detect HD root (works on any drive letter)
HD_ROOT = Path(__file__).parent

# Learning system (RAG)
def update_learning():
    """Scan lessons folder and update the vector database."""
    print("[AI] Checking for new lessons on HD...")

    from langchain_community.document_loaders import DirectoryLoader, PyPDFLoader, TextLoader
    from langchain_ollama import OllamaEmbeddings
    from langchain_chroma import Chroma
    from langchain_text_splitters import RecursiveCharacterTextSplitter

    lessons_folder = str(HD_ROOT / config["brain"]["lessons_folder"])
    memory_db = str(HD_ROOT / config["brain"]["memory_db"])

    # Load PDFs and TXTs
    loaders = [
        DirectoryLoader(lessons_folder, glob="*.pdf", loader_cls=PyPDFLoader),
        DirectoryLoader(lessons_folder, glob="*.txt", loader_cls=TextLoader),
    ]

    docs = []
    for loader in loaders:
        try:
            docs.extend(loader.load())
        except Exception:
            pass

    if not docs:
        print("[AI] No new lessons found.")
        return False

    # Split into chunks for better indexing
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=config["brain"].get("chunk_size", 1000),
        chunk_overlap=200
    )
    chunks = splitter.split_documents(docs)

    # Create/update vector database
    model_name = config["brain"].get("local_model", "llama3")
    embeddings = OllamaEmbeddings(model=model_name)

    db = Chroma.from_documents(
        documents=chunks,
        embedding=embeddings,
        persist_directory=memory_db
    )
    db.persist()
    print(f"[AI] Learning updated! {len(chunks)} chunks indexed.")
    return True


# Security filter (Security Bridge)
class SecurityBridge:
    """Controls AI access to the host PC."""

    def __init__(self, cfg):
        self.cfg = cfg

    def can_access(self, target_path: str, mode: str = "read") -> bool:
        """Check if the path is authorized in config.yaml."""
        abs_target = os.path.abspath(target_path)

        # If inside the HD itself, free access
        if abs_target.startswith(str(HD_ROOT)):
            return True

        # If on the host, check permission list
        if mode == "write" and not self.cfg["security"].get("allow_host_write", False):
            return False

        authorized = self.cfg["security"].get("allowed_folders", [])
        for path in authorized:
            if abs_target.startswith(os.path.abspath(path)):
                return True

        return False

    def safe_read(self, path: str) -> str:
        """Safely read a file from the host PC."""
        if self.can_access(path, "read"):
            with open(path, "r", encoding="utf-8") as f:
                return f.read()
        return "ACCESS DENIED: Path not authorized in config.yaml."


# Skill Loader (Plugins)
class SkillManager:
    """Manages dynamic loading of skills from the skills/ folder."""

    def __init__(self):
        self.skills = {}
        skills_path = HD_ROOT / "skills"
        if skills_path.exists():
            sys.path.insert(0, str(skills_path))
            for file in skills_path.glob("*_skill.py"):
                module_name = file.stem
                try:
                    module = importlib.import_module(module_name)
                    self.skills[module_name] = module
                    print(f"[AI] Skill loaded: {module_name}")
                except Exception as e:
                    print(f"[AI] Error loading skill {module_name}: {e}")

    def execute(self, skill_name: str, function_name: str, *args, **kwargs):
        """Execute a function inside a skill module."""
        if skill_name in self.skills:
            func = getattr(self.skills[skill_name], function_name, None)
            if func:
                return func(*args, config=config, hd_root=HD_ROOT, **kwargs)
        return f"Skill {skill_name}.{function_name} not found."


# Search context in vector database
def search_context(query: str, k: int = 3) -> str:
    """Search relevant chunks in the vector database."""
    from langchain_ollama import OllamaEmbeddings
    from langchain_chroma import Chroma

    memory_db = str(HD_ROOT / config["brain"]["memory_db"])
    model_name = config["brain"].get("local_model", "llama3")

    if not os.path.exists(memory_db):
        return ""

    embeddings = OllamaEmbeddings(model=model_name)
    db = Chroma(persist_directory=memory_db, embedding_function=embeddings)
    docs = db.similarity_search(query, k=k)

    return "\n\n".join([d.page_content for d in docs])


# Prompt execution
def ask(query: str) -> str:
    """Process user question and return AI answer."""
    mode = config["operation"]["mode"]

    # Search context in lessons
    context = search_context(query)
    prompt = f"Context from HD lessons:\n{context}\n\n---\n\nQuestion: {query}"

    if mode == "offline":
        import ollama
        model = config["brain"].get("local_model", "llama3")
        response = ollama.chat(model=model, messages=[
            {"role": "system", "content": "You are a portable AI with continuous learning. Answer based on the provided context and your general knowledge."},
            {"role": "user", "content": prompt}
        ])
        return response["message"]["content"]

    else:
        from openai import OpenAI
        api_key = config["operation"].get("api_key", "")
        if not api_key or api_key == "YOUR_KEY_HERE":
            return "ERROR: Set api_key in config.yaml to use API mode."
        client = OpenAI(api_key=api_key)
        response = client.chat.completions.create(
            model="gpt-4",
            messages=[
                {"role": "system", "content": "You are a portable AI. Use the user's lesson context to answer."},
                {"role": "user", "content": prompt}
            ]
        )
        return response.choices[0].message.content


# Command-line interface
def start_interface():
    """Interactive chat mode via terminal."""
    security = SecurityBridge(config)
    skills = SkillManager()

    print("\n" + "="*50)
    print("  PORTABLE AI - Mode: " + config["operation"]["mode"].upper())
    print("="*50)
    print("Special commands:")
    print("  /learn         - Force learning update")
    print("  /read <path>   - Read a file from host PC (if authorized)")
    print("  /skill <name> <func> <args> - Execute a skill")
    print("  /quit          - Exit")
    print("="*50 + "\n")

    while True:
        try:
            user_input = input("\nYou: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nShutting down...")
            break

        if not user_input:
            continue

        if user_input.lower() == "/quit":
            print("AI: Goodbye!")
            break

        elif user_input.lower() == "/learn":
            update_learning()

        elif user_input.lower().startswith("/read "):
            path = user_input[6:].strip()
            print("AI:", security.safe_read(path))

        elif user_input.lower().startswith("/skill "):
            parts = user_input[7:].split()
            if len(parts) >= 2:
                skill_name = parts[0]
                func_name = parts[1]
                args = parts[2:]
                result = skills.execute(skill_name, func_name, *args)
                print("AI:", result)
            else:
                print("AI: Usage: /skill <name> <function> [args]")

        else:
            answer = ask(user_input)
            print("AI:", answer)


if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="Portable AI")
    parser.add_argument("--update-db", action="store_true", help="Force vector database update")
    parser.add_argument("--interface", action="store_true", help="Start chat interface")
    args = parser.parse_args()

    if args.update_db:
        update_learning()
    else:
        update_learning()
        start_interface()

    from pathlib import Path

    def generate_image(prompt, **kwargs):
        """Generate image via Stable Diffusion (offline) or DALL-E (api)."""
        cfg = kwargs.get("config", {})
        hd_root = kwargs.get("hd_root", Path("."))
        workspace = hd_root / "workspace"

            try:
                from diffusers import StableDiffusionPipeline
                import torch
                models = hd_root / "ia_system" / "models"
                pipe = StableDiffusionPipeline.from_pretrained(
                    str(models / "stable-diffusion-v1-5")
                )
                if cfg.get("local_resources", {}).get("use_gpu", True) and torch.cuda.is_available():
                    pipe.to("cuda")
                image = pipe(prompt).images[0]
                output = workspace / "generated_image.png"
                image.save(str(output))
                return f"Image generated and saved to: {output}"
            except Exception as e:
                return f"Error generating image offline: {e}. Try API mode or install the model."
        else:
            return "API mode for image generation - configure OpenAI or Stability AI key."

HD_ROOT = Path(__file__).parent.parent
MODELS_DIR = HD_ROOT / "ia_system" / "models"
MODELS_DIR.mkdir(parents=True, exist_ok=True)

def download_model(name, repo_id, dest_subdir):
    dest = MODELS_DIR / dest_subdir
    if dest.exists():
        print(f"[OK] Model {name} already exists.")
        return
    print(f"Downloading {name}...")
    try:
        from huggingface_hub import snapshot_download
        snapshot_download(repo_id=repo_id, local_dir=str(dest))
        print(f"[OK] {name} downloaded to {dest}")
    except Exception as e:
        print(f"[ERROR] Failed to download {name}: {e}")
